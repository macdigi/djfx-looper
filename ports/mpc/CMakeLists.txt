cmake_minimum_required(VERSION 3.22)
project(DJFXMPC VERSION 1.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
option(DJFX_PRIVATE_SCOPE "Use Rare Data's private scope bridge" OFF)
option(DJFX_NATIVE_SCOPE "Build the source-edition native interface (without scopes)" OFF)
if(DJFX_PRIVATE_SCOPE AND DJFX_NATIVE_SCOPE)
    message(FATAL_ERROR "Choose the private worker or native public scopes, never both")
endif()
add_library(DJFXLooper MODULE Plugin.cpp)
if(DJFX_PRIVATE_SCOPE)
    target_sources(DJFXLooper PRIVATE ScopeTap.cpp)
endif()
target_compile_definitions(DJFXLooper PRIVATE DJFX_PRIVATE_SCOPE=$<BOOL:${DJFX_PRIVATE_SCOPE}>)
target_compile_definitions(DJFXLooper PRIVATE DJFX_NATIVE_SCOPE=$<BOOL:${DJFX_NATIVE_SCOPE}>)
set_target_properties(DJFXLooper PROPERTIES PREFIX "" SUFFIX ".so")
if(DJFX_NATIVE_SCOPE)
    set_target_properties(DJFXLooper PROPERTIES OUTPUT_NAME DJFXLooperNative)
endif()
if(APPLE)
    target_link_options(DJFXLooper PRIVATE -Wl,-exported_symbol,_VSTPluginMain)
else()
    target_link_options(DJFXLooper PRIVATE -Wl,--no-undefined -Wl,-Bsymbolic -Wl,-z,nodelete "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
endif()
if(NOT CMAKE_CROSSCOMPILING)
    add_executable(DJFXMPCContract ContractTests.cpp)
    target_link_libraries(DJFXMPCContract PRIVATE ${CMAKE_DL_LIBS})
    target_compile_definitions(DJFXMPCContract PRIVATE DJFX_PRIVATE_SCOPE=$<BOOL:${DJFX_PRIVATE_SCOPE}>)
    target_compile_definitions(DJFXMPCContract PRIVATE DJFX_NATIVE_SCOPE=$<BOOL:${DJFX_NATIVE_SCOPE}>)
    enable_testing()
    add_test(NAME StereoEffectContract COMMAND DJFXMPCContract $<TARGET_FILE:DJFXLooper>)
    if(DJFX_NATIVE_SCOPE)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME NoNativeDisplayIO COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/NoDisplayCheck.py $<TARGET_FILE:DJFXLooper>)
    endif()
endif()
