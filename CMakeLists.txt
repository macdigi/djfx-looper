cmake_minimum_required(VERSION 3.22)
project(RareDataDJFXCore VERSION 1.1.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(DJFX_NATIVE_SCOPE "Build the dedicated native MPC interface" ON)
include(CTest)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/core/ORIGIN.sha256" DJFX_ORIGIN)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/DJFXKernel.h" DJFX_KERNEL_SHA)
if(NOT DJFX_ORIGIN MATCHES "^${DJFX_KERNEL_SHA} ")
 message(FATAL_ERROR "DJFX kernel differs from its recorded source identity")
endif()
add_library(DJFXPortableCore STATIC core/DJFXLooperC.cpp)
target_include_directories(DJFXPortableCore PUBLIC core)
if(BUILD_TESTING)
 add_executable(DJFXCBridgeCheck tests/CBridgeCheck.cpp)
 target_link_libraries(DJFXCBridgeCheck PRIVATE DJFXPortableCore)
 add_test(NAME PortableCoreContract COMMAND DJFXCBridgeCheck)
endif()
add_subdirectory(ports/mpc)
